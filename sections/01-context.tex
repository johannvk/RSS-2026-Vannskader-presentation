\begin{frame}{\emph{Expert Committee on Climate Adaptation} (ECoCA)}
  \begin{itemize}
    \item Appointed by the Norwegian government, July 2024 \\[0.2cm]
    \textbf{Mandate:} Assess the socio-economic costs of climate change in Norway
    \item Concluding report presented to the Norwegian government, June 2026
  \end{itemize}
  \vspace{0.2cm}
  \textbf{Role for NR:}
  \begin{itemize}
    \item Pluvial flooding important part of Norway's
      weather/climate-related damages
      % \item Our earlier model summarised the climate over full 30-year
      %   periods
    \item Earlier experience modelling the impact of climate change on
      pluvial flooding\footcitefull{heinrich-mertsching_assessing_2023}
    \item Task: Estimate changes in pluvial flooding damages
      due to climate change
  \end{itemize}
  % \sourceNote{Expert Committee on Climate Adaptation}
\end{frame}
