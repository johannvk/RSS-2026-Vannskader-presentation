\begin{frame}{What the ratios do and do not say}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item \textbf{Relative, not absolute:} Gjensidige's portfolio
          does not cover all Norwegian buildings
        \item \textbf{Fixed portfolio:} no future construction,
          demolition or adaptation
        \item \textbf{Conservative extrapolation:} precipitation
          effects are capped at the historical training range
        \item \textbf{Uneven robustness:} national and county medians
          are stronger than municipality tail estimates
      \end{itemize}
    \end{column}
    \begin{column}{0.40\textwidth}
      \textbf{Effective ensemble size}
      \begin{itemize}
        \item Ten damage simulations based on the same climate
          projection tend to cluster
        \item The 200 ratios are therefore not 200 independent climate futures
        \item Tail quantiles should be used cautiously, especially
          for municipalities
      \end{itemize}
    \end{column}
  \end{columns}
  \sourceNote{Model assumptions and projection caveats}
\end{frame}
