
\begin{frame}{Inference framework: GAMs at scale with \texttt{mgcv}}
  \small
  \begin{equation*}
    g\!\left(\operatorname{E}[Y]\right)
    = \beta_0 + \sum_j f_j(x_j) + X\beta
  \end{equation*}

  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      % \textbf{Flexible effects}
      \begin{itemize}
        \item Allows flexible non-linear effects of covariates
          \framefootcitecompact{Wood2017}
        \item Smoothness well estimated from the data
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      % \textbf{Large-data fitting}
      \begin{itemize}
        \item \texttt{bam()} designed for large GAMs
          \framefootcitecompact{Wood2017Giga}
        \item Enables use of millions of observations
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
